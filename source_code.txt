"use client";

import { useEffect, useState } from "react";
import { supabase } from "@/lib/supabase";

export default function DashboardPage() {
  const [data, setData] = useState<any[]>([]);
  const [filtered, setFiltered] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [search, setSearch] = useState("");
  const [statusFilter, setStatusFilter] = useState("SEMUA");
  const [selected, setSelected] = useState<any>(null);

  useEffect(() => {
    loadData();
  }, []);

  useEffect(() => {
    let result = [...data];

    if (statusFilter !== "SEMUA") {
      result = result.filter(
        (item) => item.status === statusFilter
      );
    }

    if (search) {
      result = result.filter(
        (item) =>
          item.judul
            ?.toLowerCase()
            .includes(search.toLowerCase()) ||
          item.ticket_no
            ?.toLowerCase()
            .includes(search.toLowerCase())
      );
    }

    setFiltered(result);
  }, [search, statusFilter, data]);

  async function loadData() {
    setLoading(true);

    const { data, error } = await supabase
      .from("aspirations")
      .select("*")
      .eq("is_public", true)
      .order("created_at", { ascending: false });

    if (!error) {
      setData(data || []);
      setFiltered(data || []);
    }

    setLoading(false);
  }

  const total = data.length;
  const baru = data.filter(
    (x) => x.status === "BARU"
  ).length;

  const proses = data.filter(
    (x) => x.status === "DIPROSES"
  ).length;

  const klarifikasi = data.filter(
    (x) => x.status === "MENUNGGU_KLARIFIKASI"
  ).length;

  const selesai = data.filter(
    (x) => x.status === "SELESAI"
  ).length;

  const ditolak = data.filter(
    (x) => x.status === "DITOLAK"
  ).length;

  const badgeColor = (status: string) => {
    switch (status) {
      case "SELESAI":
        return "bg-green-600";

      case "DIPROSES":
        return "bg-yellow-500";

      case "DITOLAK":
        return "bg-red-600";

      case "MENUNGGU_KLARIFIKASI":
        return "bg-orange-500";

      default:
        return "bg-blue-600";
    }
  };

  return (
    <main className="min-h-screen bg-gradient-to-br from-slate-950 via-slate-900 to-slate-950 text-white p-6">
      <div className="max-w-7xl mx-auto">
        <div className="mb-8">
          <h1 className="text-5xl font-extrabold">
            📊 Dashboard Aspirasi
          </h1>

          <p className="text-slate-400 mt-3">
            Aspirasi Publik Mahasiswa Fakultas Teknik Universitas Indonesia
          </p>
        </div>

        {/* Statistik */}

        <div className="grid xl:grid-cols-6 lg:grid-cols-3 md:grid-cols-2 gap-4 mb-8">
          <div className="bg-blue-700 rounded-3xl p-6 shadow-xl">
            <div>Total Aspirasi</div>
            <div className="text-4xl font-bold mt-2">
              {total}
            </div>
          </div>

          <div className="bg-cyan-600 rounded-3xl p-6 shadow-xl">
            <div>Status Baru</div>
            <div className="text-4xl font-bold mt-2">
              {baru}
            </div>
          </div>

          <div className="bg-yellow-500 rounded-3xl p-6 shadow-xl">
            <div>Diproses</div>
            <div className="text-4xl font-bold mt-2">
              {proses}
            </div>
          </div>

          <div className="bg-orange-600 rounded-3xl p-6 shadow-xl">
            <div>Klarifikasi</div>
            <div className="text-4xl font-bold mt-2">
              {klarifikasi}
            </div>
          </div>

          <div className="bg-green-600 rounded-3xl p-6 shadow-xl">
            <div>Selesai</div>
            <div className="text-4xl font-bold mt-2">
              {selesai}
            </div>
          </div>

          <div className="bg-red-600 rounded-3xl p-6 shadow-xl">
            <div>Ditolak</div>
            <div className="text-4xl font-bold mt-2">
              {ditolak}
            </div>
          </div>
        </div>

        {/* Filter */}

        <div className="bg-slate-900 border border-slate-800 rounded-3xl p-5 mb-8">
          <div className="grid md:grid-cols-2 gap-4">
            <input
              type="text"
              placeholder="Cari nomor tiket atau judul aspirasi..."
              value={search}
              onChange={(e) => setSearch(e.target.value)}
              className="bg-slate-800 rounded-xl px-4 py-3 outline-none border border-slate-700"
            />

            <select
              value={statusFilter}
              onChange={(e) =>
                setStatusFilter(e.target.value)
              }
              className="bg-slate-800 rounded-xl px-4 py-3 border border-slate-700"
            >
              <option value="SEMUA">
                Semua Status
              </option>
              <option value="BARU">
                BARU
              </option>
              <option value="DIPROSES">
                DIPROSES
              </option>
              <option value="MENUNGGU_KLARIFIKASI">
                MENUNGGU KLARIFIKASI
              </option>
              <option value="SELESAI">
                SELESAI
              </option>
              <option value="DITOLAK">
                DITOLAK

            </select>
          </div>
        </div>

        {/* List Aspirasi */}

        {loading ? (
          <div className="bg-slate-900 rounded-3xl p-10 text-center">

        ) : (

                  📭
                </div>

                <h2 className="text-2xl font-bold">
                  Tidak Ada Data
                </h2>

                <p className="text-slate-400 mt-2">

"use client";

import { useState } from "react";
import { supabase } from "@/lib/supabase";

export default function SubmitPage() {
  const [loading, setLoading] = useState(false);
  const [showName, setShowName] = useState(false);

  const [form, setForm] = useState({
    nama: "",
    email: "",
    no_hp: "",
    jurusan: "",
    judul: "",
    isi_aspirasi: "",
  });

  const generateTicket = () => {
    const today = new Date();
    const yyyy = today.getFullYear();
    const mm = String(today.getMonth() + 1).padStart(2, "0");
    const dd = String(today.getDate()).padStart(2, "0");
    const random = Math.floor(Math.random() * 999999)
      .toString()
      .padStart(6, "0");

    return `${yyyy}${mm}${dd}-${random}`;
  };

  const handleSubmit = async (
    e: React.FormEvent<HTMLFormElement>
  ) => {
    e.preventDefault();

    setLoading(true);

    try {
      const ticketNo = generateTicket();

      const { error } = await supabase
        .from("aspirations")
        .insert([
          {
            ticket_no: ticketNo,
            nama: form.nama,
            email: form.email,
            no_hp: form.no_hp,
            jurusan: form.jurusan,
            kategori: "Umum",
            judul: form.judul,
            isi_aspirasi: form.isi_aspirasi,
            status: "BARU",
            show_name: showName,
            is_public: true,
          },
        ]);

      if (error) {
        console.error(error);
        alert("Gagal menyimpan aspirasi");
      } else {
        alert(
          `Aspirasi berhasil dikirim.\n\nNomor Tiket:\n${ticketNo}`
        );

        setForm({
          nama: "",
          email: "",
          no_hp: "",
          jurusan: "",
          judul: "",
          isi_aspirasi: "",
        });

        setShowName(false);
      }
    } catch (err) {
      console.error(err);
      alert("Terjadi kesalahan saat mengirim aspirasi");
    }

    setLoading(false);
  };

  return (
    <main className="min-h-screen bg-slate-950 p-6 md:p-10">
      <div className="max-w-4xl mx-auto">
        <div className="mb-8">

            <div className="text-4xl font-bold text-blue-400 mt-2">
              {baru}
            </div>
          </div>

          <div className="bg-slate-900 border border-slate-800 rounded-2xl p-5">
            <div className="text-slate-400">Diproses</div>
            <div className="text-4xl font-bold text-yellow-400 mt-2">
              {diproses}
            </div>
          </div>

          <div className="bg-slate-900 border border-slate-800 rounded-2xl p-5">
            <div className="text-slate-400">Selesai</div>
            <div className="text-4xl font-bold text-green-400 mt-2">
              {selesai}
            </div>
          </div>
        </div>

        {loading ? (
          <div className="bg-slate-900 rounded-2xl p-6">
            Memuat data...
          </div>
        ) : aspirasi.length === 0 ? (
          <div className="bg-slate-900 rounded-2xl p-6">
            Belum ada aspirasi yang ditampilkan.
          </div>
        ) : (
          <div className="grid gap-5">

            {aspirasi.map((item) => (
              <div
                key={item.id}
                className="bg-slate-900 border border-slate-800 rounded-2xl p-6"
              >
                <div className="flex flex-col md:flex-row md:justify-between md:items-start gap-4">

                  <div className="flex-1">
                    <h2 className="text-2xl font-semibold">
                      {item.judul}
                    </h2>

                    <p className="text-slate-400 mt-1">
                      {item.ticket_no || item.ticket_number}
                    </p>

                    <p className="text-slate-300 mt-4">
                      {item.deskripsi}
                    </p>

                    <p className="text-slate-500 mt-4 text-sm">
                      {item.jurusan}
                    </p>
                  </div>

                  <div>
                    <span
                      className={`px-4 py-2 rounded-full text-sm text-white ${statusColor(
                        item.status
                      )}`}
                    >
                      {item.status}
                    </span>
                  </div>

                </div>
              </div>
            ))}

          </div>
        )}
      </div>
    </main>
  );
}
